/**
 * 内置文件工具（P1）：路径全部限定工作空间内，错误一律转字符串回喂
 */
import { createReadStream } from "node:fs";
import fs from "node:fs/promises";
import path from "node:path";
import { z } from "zod";

export interface ToolContext {
  workspacePath: string;
  sessionId: number;
}

export interface ToolDefinition<TArgs = unknown> {
  name: string;
  description: string;
  parameters: z.ZodType<TArgs>;
  kind: "read" | "write";
  execute: (ctx: ToolContext, args: TArgs) => Promise<string>;
}

const READ_LIMIT = 512 * 1024;
const WRITE_LIMIT = 1024 * 1024;

/** 工具错误的统一文案（回喂模型，不抛出） */
function fail(message: string): string {
  return `错误: ${message}`;
}

/**
 * 解析并校验工作空间内路径：resolve 前缀校验 + realpath（symlink 跟随后再校验）
 */
export function resolveSafePath(workspacePath: string, relative: string): string {
  const resolved = path.resolve(workspacePath, relative);
  if (resolved !== workspacePath && !resolved.startsWith(workspacePath + path.sep)) {
    throw new Error("PATH_OUTSIDE_WORKSPACE");
  }
  return resolved;
}

/** 带越界校验的 realpath（文件须已存在） */
async function safeRealPath(workspacePath: string, relative: string): Promise<string> {
  const resolved = resolveSafePath(workspacePath, relative);
  const real = await fs.realpath(resolved);
  const realWs = await fs.realpath(workspacePath);
  if (real !== realWs && !real.startsWith(realWs + path.sep)) {
    throw new Error("PATH_OUTSIDE_WORKSPACE");
  }
  return real;
}

/** 简易 glob → RegExp（仅支持 * 与 ?，整段匹配文件名） */
function globToRegExp(glob: string): RegExp {
  const escaped = glob.replace(/[.+^${}()|[\]\\]/g, "\\$&").replace(/\*/g, ".*").replace(/\?/g, ".");
  return new RegExp(`^${escaped}$`);
}

const EXCLUDED_DIRS = new Set(["node_modules", ".git", "dist"]);

/** 递归收集工作空间内文件（排除清单 + 跳过二进制） */
async function collectFiles(root: string, current: string, acc: string[]): Promise<void> {
  const entries = await fs.readdir(current, { withFileTypes: true });
  for (const entry of entries) {
    if (entry.name.startsWith(".")) continue;
    if (entry.isDirectory()) {
      if (EXCLUDED_DIRS.has(entry.name)) continue;
      await collectFiles(root, path.join(current, entry.name), acc);
    } else if (entry.isFile()) {
      acc.push(path.join(current, entry.name));
    }
  }
}
```

/**
 * 审批协调器：write 工具挂起等待渲染层决议（P1 spec §1 审批挂起）
 */
type Resolver = (approved: boolean) => void;

export class ApprovalCoordinator {
  private pending = new Map<string, Resolver>();

  get pendingCount(): number {
    return this.pending.size;
  }

  request(toolCallId: string, _argSummary: string): Promise<boolean> {
    return new Promise<boolean>((resolve) => {
      this.pending.set(toolCallId, resolve);
    });
  }

  respond(toolCallId: string, approved: boolean): boolean {
    const resolve = this.pending.get(toolCallId);
    if (!resolve) {
      return false;
    }
    this.pending.delete(toolCallId);
    resolve(approved);
    return true;
  }

  /** 中断/窗口销毁：作废全部（resolve false），返回作废数量 */
  voidAll(): number {
    const count = this.pending.size;
    for (const resolve of this.pending.values()) {
      resolve(false);
    }
    this.pending.clear();
    return count;
  }
}
```

  sessionId: number,
  workspacePath: string,
  sender: WebContents | undefined,

  const set: ToolSet = {};

      parameters: def.parameters,

            const argSummary = summarizeArgs(def.name, input);
            this.emit(sender, sessionId, { type: "approval-request", toolCallId, toolName: def.name, argSummary });
            const approved = await this.approvals.request(toolCallId, argSummary);
            if (!approved) {
              return "用户拒绝了此操作";
            }
          }
        }
        this.emit(sender, sessionId, { type: "tool-update", toolCallId, toolName: def.name, state: "running" });

        });
        return output;
      },
    };
  }
